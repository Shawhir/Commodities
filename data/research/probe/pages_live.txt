    function y(v) { return m.t + (hi - v) / (hi - lo) * ih; }
    var svg = sv("svg", { viewBox: "0 0 " + W + " " + H, role: "img", "aria-label": "Daily price for the past year with 50- and 200-day averages and normal range" });
    var g = sv("g", { class: "axis" });
    niceTicks(lo, hi, narrow ? 4 : 5).forEach(function (t) {
      g.appendChild(sv("line", { x1: m.l, x2: m.l + iw, y1: y(t), y2: y(t) }));
      var tx = sv("text", { x: m.l - 8, y: y(t) + 4, "text-anchor": "end" }); tx.textContent = Math.round(t).toLocaleString("en-US"); g.appendChild(tx);
    });
    C.dates.forEach(function (d, i) { if (i && d.slice(5, 7) !== C.dates[i - 1].slice(5, 7) && (+d.slice(5, 7)) % (narrow ? 3 : 2) === 1) {
      var tx = sv("text", { x: x(i), y: H - 6, "text-anchor": "middle" }); tx.textContent = MONTHS[+d.slice(5, 7) - 1]; g.appendChild(tx); } });
    svg.appendChild(g);
    function path(a) { return a.map(function (v, i) { return v === null ? "" : (i && a[i - 1] !== null ? "L" : "M") + x(i).toFixed(1) + "," + y(v).toFixed(1); }).join(""); }
    var band = C.upper.map(function (v, i) { return v === null ? null : [x(i), y(v)]; }).filter(Boolean);
    var bandLo = C.lower.map(function (v, i) { return v === null ? null : [x(i), y(v)]; }).filter(Boolean).reverse();
    if (band.length) svg.appendChild(sv("path", { d: "M" + band.concat(bandLo).map(function (p) { return p[0].toFixed(1) + "," + p[1].toFixed(1); }).join("L") + "Z", fill: "var(--gold-soft)", stroke: "none" }));
    svg.appendChild(sv("path", { d: path(C.sma200), fill: "none", stroke: "var(--info)", "stroke-width": 1.4 }));
    svg.appendChild(sv("path", { d: path(C.sma50), fill: "none", stroke: "var(--fg)", "stroke-width": 1.2, "stroke-dasharray": "5 3", opacity: 0.6 }));
    svg.appendChild(sv("path", { d: path(C.close), fill: "none", stroke: "var(--gold)", "stroke-width": 1.8 }));
    host.appendChild(svg);
    var lg = el("div", { class: "legend" });
    [["sw", "var(--gold)", "Daily closing price"], ["sw dash", "var(--fg)", "50-day average"], ["sw", "var(--info)", "200-day average"], ["sw box", "var(--gold-soft)", "Normal daily range (20 days)"]].forEach(function (L) {
      var sp = el("span"), i = el("i", { class: L[0] });
      if (L[0] === "sw box") i.style.background = L[1]; else { i.style.borderTopColor = L[1]; if (L[0] === "sw dash") i.style.borderTopStyle = "dashed"; }
      sp.appendChild(i); sp.appendChild(document.createTextNode(L[2])); lg.appendChild(sp);
    });
    host.appendChild(lg);
    return host;
  }
  function renderTech() {

    var host = clear($("tech-body")), all = D.technical || {}, T = all[S.metal];
    if (!T) {

      $("sum-tech").textContent = all.gold ? "Gold ready; " + S.metal + " not loaded yet" : "Waiting for daily prices";
      return;
    }
    host.appendChild(jg("p", { class: "note" }, "Daily prices to " + T.as_of + ", in dollars, from " + T.first.slice(0, 4) + " (" + T.n_days.toLocaleString("en-US") + " trading days).", "Source: " + T.source));
    var p = el("div", { class: "panel" }); p.appendChild(techChart(T)); host.appendChild(p);

      top.appendChild(jg("span", { class: "tag" + (s.test.label === "useful" ? "" : " warn") }, s.test.label === "useful" ? "passed the honesty test" : s.test.n ? "for context only" : "too few cases to test",
        s.test.n ? "Walk-forward since 2010, n=" + s.test.n + " days (" + s.test.spells + " spells). Brier " + s.test.brier_setup.toFixed(3) + " vs " + s.test.brier_base.toFixed(3) +
          "; hit rate " + Math.round(s.test.hit_setup * 100) + "% vs " + Math.round(s.test.hit_base * 100) + "%." : "Too few earlier cases"));
      card.appendChild(top);
      var bars = el("div", { class: "hbars" });
      s.rates.forEach(function (r) {
        var months = r.horizon === 21 ? "1 month" : r.horizon === 63 ? "3 months" : r.horizon + " days";

      });
      card.appendChild(bars);
      var r3 = s.rates[s.rates.length - 1];
      card.appendChild(el("p", { class: "note" }, "Seen on " + r3.n_periods.toLocaleString("en-US") + " days (" + r3.n_spells + " separate spells). Blue mark: share of all days."));
      lst.appendChild(card);
    });
    if (!T.active.length) lst.appendChild(el("p", { class: "note" }, "No signal holds today."));
    host.appendChild(lst);

  }

    var hash = (location.hash || "").slice(1);
    FOLDS.forEach(function (id) { $(id).open = open.indexOf(id) >= 0 || id === hash; });
    saveOpen();
  }

     renderStretch, renderDates, renderAnalogues, renderAfter, renderTech, renderHistory, renderFoot].forEach(function (f) {
